Documentation API

Intégrez NexusAI rapidement avec nos SDKs officiels.

Python
cURL
Node.js
import nexusai client = nexusai.Client(api_key="VOTRE_API_KEY") agent = client.agents.create( name="Support Bot", model="nexus-v2", instructions="Répondre aux questions clients." ) print(f"Agent {agent.id} déployé avec succès !")
curl -X POST https://api.nexusai.com/v1/agents \ -H "Authorization: Bearer VOTRE_API_KEY" \ -H "Content-Type: application/json" \ -d '{ "name": "Support Bot", "model": "nexus-v2", "instructions": "Répondre aux questions." }'
import { NexusAI } from 'nexusai'; const client = new NexusAI('VOTRE_API_KEY'); const agent = await client.agents.create({ name: 'Support Bot', model: 'nexus-v2', instructions: 'Répondre aux questions clients.' }); console.log(`Agent ${agent.id} déployé !`);
